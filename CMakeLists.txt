# CMake build for the SKSE-Hotfixes Linux/clang-cl toolchain. Mirrors xmake.lua (upstream build stays untouched).
cmake_minimum_required(VERSION 3.21)
project(CheatCodeMenu VERSION 1.0.0 LANGUAGES CXX)
find_package(CommonLibSSE CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_commonlibsse_plugin(${PROJECT_NAME} NAME CheatCodeMenu AUTHOR cavy8 VERSION ${PROJECT_VERSION} SOURCES ${SOURCES})
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_23)
target_include_directories(${PROJECT_NAME} PRIVATE src ${CMAKE_CURRENT_SOURCE_DIR}  ${CMAKE_CURRENT_SOURCE_DIR}/ClibUtil/include ${CMAKE_CURRENT_SOURCE_DIR}/xbyak)
target_precompile_headers(${PROJECT_NAME} PRIVATE src/pch.h)
target_link_libraries(${PROJECT_NAME} PRIVATE nlohmann_json::nlohmann_json)
